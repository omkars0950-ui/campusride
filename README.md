function startTrip(id){if(!canBeRider()){showToast('Only a rider can start this trip.');return}const bookings=read(STORE_BOOKINGS,[]),b=bookings.find(x=>x.id===id),r=b&&allRides().find(x=>x.id===b.rideId);if(!b||!r||r.usn!==currentUser.usn||b.status!=='Accepted'){showToast('Only accepted trips for your own ride can be started.');return}if(!navigator.geolocation){showToast('Location is not supported by this browser.');return}showToast('Starting trip and requesting location permission…');navigator.geolocation.getCurrentPosition(pos=>{const fresh=read(STORE_BOOKINGS,[]),item=fresh.find(x=>x.id===id);if(!item)return;item.status='Started';item.location={lat:pos.coords.latitude,lng:pos.coords.longitude,accuracy:pos.coords.accuracy};item.locationUpdated=new Date().toLocaleString();item.tripStartedAt=new Date().toISOString();write(STORE_BOOKINGS,fresh);if(activeTripWatches[id]===undefined){activeTripWatches[id]=navigator.geolocation.watchPosition(p=>{const list=read(STORE_BOOKINGS,[]),entry=list.find(x=>x.id===id);if(entry&&entry.status==='Started'){entry.location={lat:p.coords.latitude,lng:p.coords.longitude,accuracy:p.coords.accuracy};entry.locationUpdated=new Date().toLocaleString();write(STORE_BOOKINGS,list)}},()=>{}, {enableHighAccuracy:true,timeout:20000,maximumAge:3000})}showToast('Trip started. Location updates are being shared while this page remains open.');renderPage()},err=>{showToast(err.code===1?'Location permission denied. Allow location and try again.':'Could not get location. Check GPS/network and try again.');},{enableHighAccuracy:true,timeout:15000,maximumAge:5000})}
function refreshTripLocation(id){
 const b=read(STORE_BOOKINGS,[]).find(x=>x.id===id);

 if(!b.location){showToast('No rider location has been shared yet.');return}

 // Cross-device live location requires a secure shared backend/realtime service.
 showToast('Showing the latest rider location saved in this browser.');renderPage();
}
function completeTrip(id){
 if(!canBeRider()){showToast('Only the rider can complete this trip.');return}
 const bookings=read(STORE_BOOKINGS,[]),b=bookings.find(x=>x.id===id),rides=allRides(),r=b&&rides.find(x=>x.id===b.rideId);
 if(!b||!r||r.usn!==currentUser.usn||b.status!=='Started'){showToast('Only your started trips can be marked complete.');return}
 if(!confirm('Mark this trip as completed? It will no longer appear as an available ride.'))return;
 b.status='Completed';b.completedAt=new Date().toISOString();
 const active=activeTripWatches[id];if(active!==undefined&&navigator.geolocation){navigator.geolocation.clearWatch(active);delete activeTripWatches[id]}
 // Once the ride reaches its destination, hide it from new partner searches.
 r.status='completed';write(STORE_BOOKINGS,bookings);write(STORE_RIDES,rides);
 showToast('Trip completed. The ride is no longer available for new requests.');renderPage();
}

function cancelBooking(id){const bookings=read(STORE_BOOKINGS,[]),b=bookings.find(x=>x.id===id);if(!b||b.status==='Cancelled')return;const wasAccepted=b.status==='Accepted';b.status='Cancelled';const rides=allRides(),r=rides.find(x=>x.id===b.rideId);if(r&&wasAccepted)r.seats++;write(STORE_BOOKINGS,bookings);write(STORE_RIDES,rides);showToast('Seat request cancelled.');renderPage()}
function cancelRide(id){if(!confirm('Remove this ride offer?'))return;write(STORE_RIDES,allRides().filter(r=>r.id!==id));showToast('Ride offer removed.');renderPage()}

function locationError(err){
 const status=$('locationStatus');if(!status)return;
 const messages={1:'Location permission was denied. Allow location access in your browser settings and try again.',2:'Your location could not be determined. Check GPS or network access and try again.',3:'Location request timed out. Please try again.'};
 status.textContent=messages[err.code]||'Could not access your location. Please try again.';
 status.style.background='#fff3ed';status.style.color='#a34b1a';
}
function locationSuccess(pos){
 lastCoordinates={latitude:pos.coords.latitude,longitude:pos.coords.longitude,accuracy:pos.coords.accuracy};
 const {latitude,longitude,accuracy}=lastCoordinates;
 const status=$('locationStatus');
 if(status){status.style.background='#effaf5';status.style.color='#18744e';status.innerHTML=`<b>Location found</b><br>Latitude: ${latitude.toFixed(6)} · Longitude: ${longitude.toFixed(6)}<br>Approximate accuracy: ${Math.round(accuracy)} metres · Updated: ${new Date().toLocaleTimeString()}`}
 updateMapLocation(latitude,longitude);
}
function getMyLocation(){
 if(!navigator.geolocation){showToast('This browser does not support location access.');return}
 const s=$('locationStatus');if(s){s.textContent='Requesting your location…';s.style.background='#f3f7fc';s.style.color='#53657b'}
 navigator.geolocation.getCurrentPosition(locationSuccess,locationError,{enableHighAccuracy:true,timeout:15000,maximumAge:5000});
}
function startLocationTracking(){
 if(!navigator.geolocation){showToast('This browser does not support location access.');return}
 if(locationWatchId!==null){showToast('Live location tracking is already on.');return}
 const s=$('locationStatus');if(s)s.textContent='Starting live location updates…';
 locationWatchId=navigator.geolocation.watchPosition(locationSuccess,locationError,{enableHighAccuracy:true,timeout:20000,maximumAge:3000});
 showToast('Live location updates started. Keep this page open.');
}
function stopLocationTracking(){
 if(locationWatchId!==null&&navigator.geolocation){navigator.geolocation.clearWatch(locationWatchId);locationWatchId=null;showToast('Live location updates stopped.')}
}
function updateMapLocation(lat,lng){
 const frame=$('googleMapFrame');if(frame)frame.src=`https://maps.google.com/maps?q=${encodeURIComponent(lat+','+lng)}&z=16&output=embed`;
}
function openGoogleDirections(){
 const from=$('mapPickup')?.value.trim()||'';
 const to=$('mapDestination')?.value.trim()||'';
 let url;
 if(lastCoordinates&&!from){url=`https://www.google.com/maps/dir/?api=1&origin=${encodeURIComponent(lastCoordinates.latitude+','+lastCoordinates.longitude)}${to?'&destination='+encodeURIComponent(to):''}&travelmode=driving`}
 else if(from&&to){url=`https://www.google.com/maps/dir/?api=1&origin=${encodeURIComponent(from)}&destination=${encodeURIComponent(to)}&travelmode=driving`}
 else if(to){url=`https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(to)}`}
 else {showToast('Enter a pickup and destination, or get your current location first.');return}
 window.open(url,'_blank','noopener,noreferrer');
}
async function shareLocation(){
 if(!lastCoordinates){showToast('Get your current location first.');return}
 const url=`https://www.google.com/maps?q=${lastCoordinates.latitude},${lastCoordinates.longitude}`;
 const data={title:'My CampusRide location',text:'Here is my current location:',url};
 try{
   if(navigator.share){await navigator.share(data);showToast('Location sharing opened.')}
   else if(navigator.clipboard&&window.isSecureContext){await navigator.clipboard.writeText(url);showToast('Google Maps location link copied to clipboard.')}
   else {window.prompt('Copy this Google Maps location link:',url)}
 }catch(err){if(err.name!=='AbortError')showToast('Could not open sharing. You can copy the Google Maps link manually.')}
}
